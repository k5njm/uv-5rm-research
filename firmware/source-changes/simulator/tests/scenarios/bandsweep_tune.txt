# Band Sweep: after one pass MENU listens to the selected hit; VFO/MR puts that frequency on the main screen's VFO and leaves.
master 146000000
signal 146520000 120 -1 always
app bandsweep
wait 7
shot hits
key menu
shot listening
expect noexit
expect vfo none
key vm
expect exit
expect vfo 146512500 0     # 2 m sweeps at 12.5 kHz: the signal at 146.520 falls between two grid points, both read full strength
